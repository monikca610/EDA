Program 5: Scatter Plot – Age vs Experience:

# Program 5: Scatter Plot

import pandas as pd
import matplotlib.pyplot as plt

# Load the Employee dataset
df = pd.read_csv("Employee (3)(1).csv")

# Create Scatter Plot
plt.figure(figsize=(8,5))

plt.scatter(df["Age"], df["ExperienceInCurrentDomain"])

plt.title("Age vs Experience in Current Domain")
plt.xlabel("Age")
plt.ylabel("Experience in Current Domain")

plt.grid(True)
plt.show()
