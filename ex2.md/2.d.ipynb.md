Program 4: Box Plot – Employee Age:

# Program 4: Box Plot

import pandas as pd
import matplotlib.pyplot as plt

# Load the Employee dataset
df = pd.read_csv("Employee (3)(1).csv")

# Create Box Plot
plt.figure(figsize=(6,5))

plt.boxplot(df["Age"])

plt.title("Box Plot of Employee Age")
plt.ylabel("Age")

plt.grid(True)
plt.show()
