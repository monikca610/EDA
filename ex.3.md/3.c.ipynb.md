Program 3: Replace Missing Numerical Values Using Mean:

import pandas as pd

# Load the Employee dataset
df = pd.read_csv("Employee.csv")

# Fill missing numerical values with mean
df["JoiningYear"] = df["JoiningYear"].fillna(
    df["JoiningYear"].mean()
)

df["PaymentTier"] = df["PaymentTier"].fillna(
    df["PaymentTier"].mean()
)

df["Age"] = df["Age"].fillna(
    df["Age"].mean()
)

df["ExperienceInCurrentDomain"] = df[
    "ExperienceInCurrentDomain"
].fillna(
    df["ExperienceInCurrentDomain"].mean()
)

print(df)
