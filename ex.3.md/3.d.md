Program 4: Replace Missing Categorical Values Using Mode:

import pandas as pd

# Load the Employee dataset
df = pd.read_csv("Employee.csv")

# Fill missing categorical values with mode
df["Education"] = df["Education"].fillna(
    df["Education"].mode()[0]
)

df["City"] = df["City"].fillna(
    df["City"].mode()[0]
)

df["Gender"] = df["Gender"].fillna(
    df["Gender"].mode()[0]
)

df["EverBenched"] = df["EverBenched"].fillna(
    df["EverBenched"].mode()[0]
)

print(df)
