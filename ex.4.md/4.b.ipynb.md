Program 2: Convert Data Types:

import pandas as pd

# Load the Employee dataset
df = pd.read_csv("Employee (3).csv")

# Rename columns
df.rename(columns={
    "JoiningYear": "Joining_Year",
    "PaymentTier": "Payment_Tier",
    "Age": "Age",
    "ExperienceInCurrentDomain": "Domain_Experience",
    "LeaveOrNot": "Leave_Status"
}, inplace=True)

# Display data types before conversion
print("Before Conversion:")
print(df.dtypes)

# Convert numerical columns
df["Joining_Year"] = df["Joining_Year"].astype(int)
df["Payment_Tier"] = df["Payment_Tier"].astype(int)
df["Age"] = df["Age"].astype(int)
df["Domain_Experience"] = df["Domain_Experience"].astype(int)
df["Leave_Status"] = df["Leave_Status"].astype(int)
