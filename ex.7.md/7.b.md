Program 2: Up-Sampling (Monthly to Daily) :

import pandas as pd

# Load the Employee dataset
df = pd.read_csv("Employee.csv")

# Create Date column from JoiningYear
df["Date"] = pd.to_datetime(
    df["JoiningYear"].astype(str) + "-01-01"
)

# Set Date as index
df.set_index("Date", inplace=True)

# Down-Sampling: Yearly summary
yearly_data = df.resample("YE").agg({
    "Age": "mean",
    "PaymentTier": "mean",
    "ExperienceInCurrentDomain": "mean",
    "LeaveOrNot": "sum"
})

# Up-Sampling: Yearly to Monthly
monthly_data = yearly_data.resample("ME").ffill()
