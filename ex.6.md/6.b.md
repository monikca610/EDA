Program 2: Construct Pair Plot with Categorical Hue:

import pandas as pd
import seaborn as sns
import matplotlib.pyplot as plt

# Load the Employee dataset
df = pd.read_csv("Employee (3).csv")

# Create Pair Plot
sns.pairplot(
    df,
    vars=[
        "JoiningYear",
        "PaymentTier",
        "Age",
        "ExperienceInCurrentDomain",
        "LeaveOrNot"
    ],
    hue="Gender",
    diag_kind="hist"
)

# Display the plot
plt.show()
